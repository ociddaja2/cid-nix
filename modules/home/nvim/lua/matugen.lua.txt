    base00 = '#000000',
    base01 = '#1f0f0c',
    base02 = '#2b1916',
    base03 = '#ae8882',
    base04 = '#e7bdb6',
    base05 = '#fddbd6',
    base06 = '#fddbd6',
    base07 = '#fddbd6',
    base08 = '#ffb4ab',
    base09 = '#ffb951',
    base0A = '#ffb4a8',
    base0B = '#ffb4a8',
    base0C = '#ffb951',
    base0D = '#ffb4a8',
    base0E = '#ffb4a8',
    base0F = '#ffdad5',
  })

  local hi = function(group, opts)
    vim.api.nvim_set_hl(0, group, opts)
  end

  -- telescope.nvim
  hi('TelescopeNormal',         { fg = '#fddbd6',          bg = '#000000' })
  hi('TelescopeBorder',         { fg = '#ae8882',             bg = '#000000' })
  hi('TelescopePromptNormal',   { fg = '#fddbd6',          bg = '#000000' })
  hi('TelescopePromptBorder',   { fg = '#ae8882',             bg = '#000000' })
  hi('TelescopePromptPrefix',   { fg = '#ffb4a8',             bg = '#000000' })
  hi('TelescopePromptCounter',  { fg = '#e7bdb6',  bg = '#000000' })
  hi('TelescopePromptTitle',    { fg = '#000000',             bg = '#ffb4a8' })
  hi('TelescopePreviewTitle',   { fg = '#000000',             bg = '#ffb4a8' })
  hi('TelescopeResultsTitle',   { fg = '#000000',             bg = '#ffb951' })
  hi('TelescopeSelection',      { fg = '#fddbd6',          bg = '#2b1916' })
  hi('TelescopeSelectionCaret', { fg = '#ffb4a8',             bg = '#2b1916' })
  hi('TelescopeMatching',       { fg = '#ffb4a8',             bold = true })

  -- mini.pick
  hi('MiniPickNormal',         { fg = '#fddbd6',          bg = '#000000' })

-- Register a signal handler for SIGUSR1 (matugen updates).
-- The handler re-requires this module, which re-runs the code below, so the
-- previous handle is stopped first; otherwise handlers double on every signal.
if _G.__matugen_signal then
  _G.__matugen_signal:stop()
  _G.__matugen_signal:close()
end

local signal = vim.uv.new_signal()
_G.__matugen_signal = signal
signal:start(
  'sigusr1',
  vim.schedule_wrap(function()
    package.loaded['matugen'] = nil
    require('matugen').setup()
  end)
)

return M
